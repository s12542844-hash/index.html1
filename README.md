    e.cipher.data = b64(new Uint8Array(encrypted));
    payload.fill(0);
    return e;
  }
  async function decrypt(e,key) {

      bytes = new Uint8Array(await window.crypto.subtle.decrypt({name:'AES-GCM',iv:unb64(e.cipher.iv,'متجه التشفير',12,12),additionalData:aad(e),tagLength:128},key,unb64(e.cipher.data,'البيانات',16,MAX_TEXT)));
    } catch (error) { fail('كلمة المرور غير صحيحة أو النسخة المشفرة تالفة.', 'AUTH'); }

    try { payload=parseJSON(decoder.decode(bytes)); } finally { bytes.fill(0); }

    requireCurrent(ref);

      try {

      } catch(ignore) {}
      ref.key=key; ref.salt=salt; ref.vault=vault; ref.revision=e.revision; ref.raw=raw;
      return snapshot();
    });

        return r;
      });
    }
    return validateVault({schemaVersion:VERSION,currentBusinessId:primary.id,businesses:Array.from(map.values())});
  }

  });

  // Small mathematical grammar: numbers, parentheses, arithmetic and postfix percent only.
  // No JavaScript evaluation, functions, object access or assignment are accepted.
  function arithmetic(expression) {
    const source = ascii(expression).replace(/×/g, '*').replace(/÷/g, '/').replace(/−/g, '-').replace(/\s+/g, '');
    if (!source || source.length > 160 || /[^\d.+*/()%\-]/.test(source)) return null;
    let pos = 0, depth = 0;
    function factor() {
      if (++depth > 20) throw new Error('depth');
      let sign = 1;
      if (source[pos] === '+' || source[pos] === '-') sign = source[pos++] === '-' ? -1 : 1;
      let value;
      if (source[pos] === '(') {
        pos++; value = sum();
        if (source[pos++] !== ')') throw new Error('parenthesis');
      } else {
        const m = source.slice(pos).match(/^(?:\d+(?:\.\d*)?|\.\d+)/);
        if (!m) throw new Error('number');
        value = Number(m[0]); pos += m[0].length;
      }
      if (source[pos] === '%') { value /= 100; pos++; }
      depth--; return value * sign;
    }
    function product() {
      let value = factor();
      while (source[pos] === '*' || source[pos] === '/') {
        const op = source[pos++], operand = factor();
        if (op === '/' && operand === 0) throw new Error('zero');
        value = op === '*' ? value * operand : value / operand;
      }
      return value;
    }
    function sum() {
      let value = product();
      while (source[pos] === '+' || source[pos] === '-') {
        const op = source[pos++], operand = product();
        value = op === '+' ? value + operand : value - operand;
      }
      return value;
    }
    try {
      const result = sum();
      return pos === source.length && Number.isFinite(result) && Math.abs(result) <= 1e15 ? result : null;
    } catch (_) { return null; }
  }
